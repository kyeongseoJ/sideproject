```
cd C:\Users\6122\Desktop\DXSchool_JKS\JAVA\vibecoding\sideproject

.\back\mvnw.cmd -DskipTests compile

$classes = (Resolve-Path ".\back\target\classes").Path
$spring = Get-ChildItem "$env:USERPROFILE\.m2\repository\org\springframework\spring-context" `
  -Recurse -Filter "spring-context-*.jar" |
  Sort-Object LastWriteTime |
  Select-Object -Last 1 -ExpandProperty FullName

$harness = "$env:TEMP\NoveltyMissionRecommendationSimulation.java"

javac -encoding UTF-8 `
  -cp "$classes;$spring" `
  -d $env:TEMP `
  $harness

java "-Dfile.encoding=UTF-8" `
  -cp "$env:TEMP;$classes;$spring" `
  NoveltyMissionRecommendationSimulation

```
